datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

enum Role {
  STUDENT
  ADMIN
}

model User {
  id           String        @id @default(uuid())
  name         String
  email        String        @unique
  password     String
  role         Role          @default(STUDENT)
  skills       String[]      @default([])
  createdAt    DateTime      @default(now())
  updatedAt    DateTime      @updatedAt
  enrollments  Enrollment[]
  projects     Project[]
  certificates Certificate[]
  offerLetters OfferLetter[]
}

model Course {
  id          String       @id @default(uuid())
  title       String
  category    String // e.g. "Programming", "Web Development", "AI & Data", "Cyber Security", "Internship"
  description String
  duration    String // e.g. "8 Weeks", "3 Months"
  type        String       @default("COURSE") // "COURSE" or "INTERNSHIP"
  skills      String[]     @default([]) // relevant skills taught
  lessons     Json[]       @default([]) // stores array of lessons: { title: string, videoUrl: string, duration: string }
  assignments Json[]       @default([]) // stores assignments: { id: string, title: string, desc: string }
  quizzes     Json[]       @default([]) // stores quizzes: { question: string, options: string[], answer: string }
  createdAt   DateTime     @default(now())
  updatedAt   DateTime     @updatedAt
  enrollments Enrollment[]
}

model Enrollment {
  id          String   @id @default(uuid())
  userId      String
  user        User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  courseId    String
  course      Course   @relation(fields: [courseId], references: [id], onDelete: Cascade)
  progress    Int      @default(0) // 0 to 100
  status      String   @default("ENROLLED") // ENROLLED, COMPLETED
  linkedinUrl String?  // Stores LinkedIn verification link if submitted
  joinedAt    DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@unique([userId, courseId])
}

model Project {
  id          String   @id @default(uuid())
  studentId   String
  student     User     @relation(fields: [studentId], references: [id], onDelete: Cascade)
  title       String   // Project / Internship Title
  description String
  githubLink  String
  fileUrl     String?  // Optional uploaded file URL/path
  status      String   @default("PENDING") // PENDING, APPROVED, REJECTED
  feedback    String?
  submittedAt DateTime @default(now())
  updatedAt   DateTime @updatedAt
}

model Certificate {
  id                String   @id @default(uuid())
  studentId         String
  student           User     @relation(fields: [studentId], references: [id], onDelete: Cascade)
  courseName        String   // web development internship, python developer internship
  certificateNumber String   @unique // e.g. VINIX-WEB-2026-0001
  issueDate         DateTime @default(now())
  verificationURL   String
}

model OfferLetter {
  id                String   @id @default(uuid())
  offerLetterId     String   @unique // e.g. VINIX-OFFER-2026-0001
  studentId         String
  student           User     @relation(fields: [studentId], references: [id], onDelete: Cascade)
  internshipId      String?  
  studentName       String
  studentEmail      String
  internshipTitle   String
  internshipDomain  String
  startDate         DateTime
  endDate           DateTime
  duration          String
  mentorName        String
  issueDate         DateTime @default(now())
  status            String   @default("GENERATED") // GENERATED, SENT, ACCEPTED, DECLINED, EXPIRED
  pdfUrl            String?
  verificationToken String   @unique @default(uuid())
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt
}

model InternshipApplication {
  id                  String    @id @default(dbgenerated("gen_random_uuid()")) @db.Uuid
  userId              String    @map("user_id") @db.Uuid
  internshipId        String    @map("internship_id")
  status              String?   @default("active")
  appliedAt           DateTime? @default(now()) @map("applied_at") @db.Timestamptz
  updatedAt           DateTime? @default(now()) @map("updated_at") @db.Timestamptz
  studentName         String    @map("student_name")
  email               String
  phone               String?
  college             String?
  domain              String?
  internshipName      String?   @map("internship_name")
  startDate           DateTime? @map("start_date") @db.Timestamptz
  endDate             DateTime? @map("end_date") @db.Timestamptz
  certificateStatus   String?   @default("pending") @map("certificate_status")
  offerLetterStatus   String?   @default("pending") @map("offer_letter_status")
  progress            Int?      @default(0)
  mentorId            String?   @map("mentor_id") @db.Uuid

  @@unique([userId, internshipId])
  @@map("internship_applications")
}
